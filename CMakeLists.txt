cmake_minimum_required(VERSION 3.16)
project(qpack CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(ANDROID)
  set(_native_default OFF)
else()
  set(_native_default ON)
endif()
option(QPACK_NATIVE "optimise for the build machine (-march=native)" ${_native_default})
option(QPACK_BUILD_TOOLS "build the command line tools and tests" ON)
if(QPACK_NATIVE)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-march=native>)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
  # portable AArch64 baseline: ARMv8.2 with the dot-product and fp16 extensions (Cortex-A55 / A76 and newer: phones, Galaxy Watch 8)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-march=armv8.2-a+dotprod+fp16>)
endif()
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-O3> $<$<COMPILE_LANGUAGE:CXX>:-Wall> $<$<COMPILE_LANGUAGE:CXX>:-Wextra> $<$<COMPILE_LANGUAGE:CXX>:-Wno-unused-parameter>)
find_package(Threads REQUIRED)

option(QPACK_CUDA "build the CUDA backend (QPACK_DEVICE=cuda; Llama family, Pascal and newer)" OFF)
set(QPACK_CUDA_ARCH "61" CACHE STRING "CUDA architectures for the CUDA backend")
if(QPACK_CUDA)
  set(CMAKE_CUDA_ARCHITECTURES ${QPACK_CUDA_ARCH})
  set(CMAKE_CUDA_STANDARD 17)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_compile_options($<$<COMPILE_LANGUAGE:CUDA>:-O3> $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()

option(QPACK_SIZE_OPT "optimise everything except the kernels for size (-Os, LTO, section GC); the kernels stay -O3" OFF)
if(QPACK_SIZE_OPT)
  add_compile_options(-ffunction-sections -fdata-sections -flto)
  if(APPLE)
    add_link_options(-flto -Wl,-dead_strip)
  else()
    add_link_options(-flto -Wl,--gc-sections)
  endif()
endif()

add_library(qpack STATIC src/json.cpp src/container.cpp src/kernels.cpp src/kernels_x86.cpp src/kernels_neon.cpp src/threads.cpp src/model_llama.cpp src/model_lfm.cpp src/model_gptoss.cpp src/model_nq.cpp src/tokenizer.cpp src/unicode.cpp src/sampler.cpp src/chat.cpp src/circuits.cpp)
if(QPACK_SIZE_OPT)
  file(GLOB _cold src/*.cpp)
  list(FILTER _cold EXCLUDE REGEX "kernels|model_|sampler|threads")   # hot paths stay -O3
  set_source_files_properties(${_cold} PROPERTIES COMPILE_OPTIONS "-Os")
endif()
target_include_directories(qpack PUBLIC include)
target_link_libraries(qpack PUBLIC Threads::Threads)
if(QPACK_CUDA)
  target_sources(qpack PRIVATE src/model_llama_cuda.cu src/model_gptoss_cuda.cu)
  target_compile_definitions(qpack PUBLIC QPACK_HAVE_CUDA=1)
  target_link_libraries(qpack PUBLIC CUDA::cudart_static)
endif()

if(QPACK_BUILD_TOOLS)
foreach(tool qpack-cli qpack-server qpack-bench qpack-quantize qpack-perplexity)
  add_executable(${tool} tools/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE qpack)
endforeach()

if(QPACK_CUDA)
  foreach(tool qpack-cudacheck qpack-cudamv)
    add_executable(${tool} tools/${tool}.cpp)
    target_link_libraries(${tool} PRIVATE qpack)
  endforeach()
  add_executable(cuda_dump bench/cuda_dump.cpp)
  target_link_libraries(cuda_dump PRIVATE qpack)
endif()

add_executable(test_tokenizer tests/test_tokenizer.cpp)
target_link_libraries(test_tokenizer PRIVATE qpack)
add_executable(test_circuits tests/test_circuits.cpp)
target_link_libraries(test_circuits PRIVATE qpack)
endif()

if(ANDROID)
  add_subdirectory(android/jni)
endif()
