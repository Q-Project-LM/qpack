cmake_minimum_required(VERSION 3.16)
project(qpack CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(QPACK_NATIVE "optimise for the build machine (-march=native)" ON)
if(QPACK_NATIVE)
  add_compile_options(-march=native)
endif()
add_compile_options(-O3 -Wall -Wextra -Wno-unused-parameter)
find_package(Threads REQUIRED)

add_library(qpack STATIC src/json.cpp src/container.cpp src/kernels.cpp src/threads.cpp src/model_llama.cpp src/model_nq.cpp src/tokenizer.cpp src/unicode.cpp src/sampler.cpp src/chat.cpp src/circuits.cpp)
target_include_directories(qpack PUBLIC include)
target_link_libraries(qpack PUBLIC Threads::Threads)

foreach(tool qpack-cli qpack-server qpack-bench qpack-quantize qpack-perplexity)
  add_executable(${tool} tools/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE qpack)
endforeach()

add_executable(test_tokenizer tests/test_tokenizer.cpp)
target_link_libraries(test_tokenizer PRIVATE qpack)
add_executable(test_circuits tests/test_circuits.cpp)
target_link_libraries(test_circuits PRIVATE qpack)
